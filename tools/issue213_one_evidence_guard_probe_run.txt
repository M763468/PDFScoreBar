from __future__ import annotations

import json
import re
from collections import Counter
from pathlib import Path
from typing import Any

import cv2
import torch

from src.measure_numbering.mmr import MMRClassifier, MMROCREngine
from src.measure_numbering.rapidocr_provider import create_mmr_rapidocr
from tools.issue94.eval_mmr_overrides import _build_summary, _load_json

ROOT = Path(__file__).resolve().parents[1]
PAGE_INPUTS = ROOT / "logs" / "issue94_mmr_current_state" / "page_inputs.json"
MODEL_PATH = ROOT / "tools" / "mmr_training" / "models" / "mmr_classifier_best.pth"
OUT_DIR = ROOT / "logs" / "issue213_one_evidence_guard_probe"
REPORT = OUT_DIR / "issue213_guard_probe_report.txt"
DETAIL_JSON = OUT_DIR / "issue213_guard_probe_details.json"

THRESHOLD = 0.5
RESCUE_THRESHOLD = 0.1
PROBE_PROB_MAX = 0.60
PROBE_SCORE_MAX = 0.0
ONE_EVIDENCE_MIN_COUNT = 2
ONE_EVIDENCE_MIN_CONF = 0.80


def j(value: Any) -> str:
    return json.dumps(value, indent=2, ensure_ascii=False)


def jsonable(value: Any) -> Any:
    if value is None or isinstance(value, (str, int, float, bool)):
        return value
    if isinstance(value, Path):
        return str(value)
    if isinstance(value, dict):
        return {str(k): jsonable(v) for k, v in value.items()}
    if isinstance(value, (list, tuple)):
        return [jsonable(v) for v in value]
    try:
        import numpy as np

        if isinstance(value, np.generic):
            return value.item()
        if isinstance(value, np.ndarray):
            return value.tolist()
    except Exception:
        pass
    return repr(value)


def aggregate(page_summaries: list[dict[str, Any]]) -> dict[str, Any]:
    total_pages = len(page_summaries)
    total_base = sum(p["counts"]["base_measures"] for p in page_summaries)
    total_expected = sum(p["counts"]["expected_overrides"] for p in page_summaries)
    total_detected = sum(p["counts"]["detected_overrides"] for p in page_summaries)
    total_matched = sum(p["counts"]["matched"] for p in page_summaries)
    total_missed = sum(p["counts"]["missed"] for p in page_summaries)
    total_mismatch = sum(p["counts"]["skip_mismatch"] for p in page_summaries)
    total_unexpected = sum(p["counts"]["unexpected"] for p in page_summaries)
    precision = total_matched / total_detected if total_detected else 0.0
    recall = total_matched / total_expected if total_expected else 0.0
    f1 = 2 * precision * recall / (precision + recall) if precision + recall else 0.0
    return {
        "total_pages": total_pages,
        "total_base_measures": total_base,
        "total_expected": total_expected,
        "total_detected": total_detected,
        "matched_tp": total_matched,
        "missed_fn": total_missed,
        "skip_mismatch": total_mismatch,
        "unexpected_fp": total_unexpected,
        "precision": precision,
        "recall": recall,
        "f1_score": f1,
    }


def score_candidates_with_one_evidence(
    ocr: MMROCREngine, ocr_result: list[Any] | None, img_width: int, img_height: int
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    if not ocr_result:
        return [], []
    all_mmr_candidates: list[dict[str, Any]] = []
    one_candidates: list[dict[str, Any]] = []
    center_x = img_width / 2.0

    for item, source in ocr._candidate_items(ocr_result):
        box_points, text, conf = item
        clean_text = re.sub(r"^[EP](\d)", r"\1", text)
        clean_text = re.sub(r"[.,;]", "", clean_text)
        blacklisted = ocr._has_blacklisted_text(text)
        nums_found = ocr._extract_numeric_candidates(clean_text, blacklisted)
        if not nums_found:
            continue

        xs = [p[0] for p in box_points]
        ys = [p[1] for p in box_points]
        x_min, x_max, y_min, y_max = min(xs), max(xs), min(ys), max(ys)
        box_h = y_max - y_min
        box_center_x = (x_min + x_max) / 2.0
        box_center_y = (y_min + y_max) / 2.0
        dist_x_norm = abs(box_center_x - center_x) / img_width
        dist_y_norm = abs(box_center_y - (img_height / 2.0)) / img_height
        h_ratio = box_h / img_height

        for n_str in nums_found:
            try:
                val = int(n_str)
            except ValueError:
                continue

            score = 100 - dist_x_norm * 200 - dist_y_norm * 100
            if 0.4 <= h_ratio <= 0.95:
                score += 20
            elif h_ratio < 0.3:
                score -= 30
            if "=" in text:
                parts = text.split("=")
                if len(parts) > 1 and n_str in parts[1]:
                    score -= 80
            if val > 100:
                score -= 50
            if val > 20 and img_width < 100:
                score -= 200

            rec = {
                "val": val,
                "score": score,
                "text": text,
                "conf": float(conf),
                "debug": f"dx={dist_x_norm:.2f},dy={dist_y_norm:.2f},h={h_ratio:.2f},{source}",
                "source": source,
            }
            if val == 1:
                one_candidates.append(rec)
            elif val >= 2:
                all_mmr_candidates.append(rec)
    return all_mmr_candidates, one_candidates


def detect_with_probe(
    image,
    system: dict[str, Any],
    x1: float,
    y1: float,
    x2: float,
    y2: float,
    prob: float,
    w_img: int,
    h_img: int,
    ocr: MMROCREngine,
) -> dict[str, Any]:
    variants = [("standard", 0)]
    if prob > THRESHOLD:
        variants = [("standard", 0), ("no_dilate", 0), ("heavy_dilate", 0)]

    variant_results: list[tuple[int, float, str]] = []
    one_evidence: list[dict[str, Any]] = []
    variant_details: list[dict[str, Any]] = []

    for mode, angle in variants:
        stave_results: list[tuple[int | None, float, str]] = []
        stave_details: list[dict[str, Any]] = []
        for stave_idx, stave in enumerate(system.get("staves", [])):
            s_bbox = stave["bbox"]
            margin_y = 80
            ox1, ox2 = int(max(0, x1 - 30)), int(min(w_img, x2 + 30))
            oy1, oy2 = int(max(0, s_bbox[1] - margin_y)), int(min(h_img, s_bbox[3] + 80))
            stave_crop = image[oy1:oy2, ox1:ox2]
            stave_crop = ocr.mask_hbar_candidates(stave_crop, margin_y, s_bbox[3] - s_bbox[1])
            proc_img = ocr.preprocess_variant(stave_crop, mode=mode, angle=angle)
            if proc_img is None:
                continue
            raw_ocr, _ = ocr.ocr_engine(proc_img)
            mmr_candidates, ones = score_candidates_with_one_evidence(ocr, raw_ocr, ox2 - ox1, oy2 - oy1)
            for one in ones:
                one_evidence.append({"variant": f"{mode}:{angle}", "stave_idx": stave_idx, **one})
            if mmr_candidates:
                best = sorted(mmr_candidates, key=lambda x: x["score"], reverse=True)[0]
                stave_results.append((best["val"], best["score"], best["debug"]))
                selected = best
            else:
                stave_results.append((None, 0.0, ""))
                selected = None
            stave_details.append(
                {
                    "stave_idx": stave_idx,
                    "raw_ocr_repr": repr(raw_ocr),
                    "selected_mmr_candidate": selected,
                    "one_candidates": ones,
                }
            )

        valid_results = [r for r in stave_results if r[0] is not None]
        if valid_results:
            counts = Counter([r[0] for r in valid_results])
            current_num = counts.most_common(1)[0][0]
            if not (current_num > 20 and (x2 - x1) < 100):
                _, best_score, best_dbg = max(
                    [r for r in valid_results if r[0] == current_num], key=lambda x: x[1]
                )
                variant_results.append((current_num, best_score, f"{best_dbg},variant={mode}:{angle}"))
        variant_details.append({"variant": f"{mode}:{angle}", "staves": stave_details})

    if variant_results:
        found_num, best_score, best_debug = max(variant_results, key=lambda x: x[1])
    else:
        found_num, best_score, best_debug = None, 0.0, ""

    strong_one = [o for o in one_evidence if o["conf"] >= ONE_EVIDENCE_MIN_CONF]
    proposed_veto = bool(
        found_num
        and prob > THRESHOLD
        and prob < PROBE_PROB_MAX
        and best_score < PROBE_SCORE_MAX
        and len(strong_one) >= ONE_EVIDENCE_MIN_COUNT
    )
    return {
        "found_num": found_num,
        "best_score": best_score,
        "best_debug": best_debug,
        "one_evidence": one_evidence,
        "strong_one_count": len(strong_one),
        "strong_one_max_conf": max([o["conf"] for o in strong_one], default=0.0),
        "proposed_veto": proposed_veto,
        "variant_details": variant_details,
    }


def build_eval_summary(page_input, numbering_payload, overrides):
    expected_path = ROOT / "tests" / "fixtures" / f"expected_overrides_{page_input['page_id']}.json"
    expected_payload = _load_json(expected_path) if expected_path.exists() else None
    return _build_summary(
        numbering_json=ROOT / page_input["numbering_base"],
        image=ROOT / page_input["image"],
        model_path=MODEL_PATH,
        rapidocr_provider="cuda",
        threshold=THRESHOLD,
        rescue_threshold=RESCUE_THRESHOLD,
        enable_rotation_tta=False,
        numbering_payload=numbering_payload,
        detected_payload={"measure_overrides": overrides},
        expected_payload=expected_payload,
    )


def main() -> None:
    OUT_DIR.mkdir(parents=True, exist_ok=True)
    inputs = json.loads(PAGE_INPUTS.read_text(encoding="utf-8"))
    pages = inputs["pages"]
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    classifier = MMRClassifier(MODEL_PATH, device)
    rapidocr = create_mmr_rapidocr("cuda")
    ocr = MMROCREngine(enable_rotation_tta=False, ocr_engine=rapidocr)

    current_summaries = []
    proposed_summaries = []
    current_records = []
    vetoed_records = []
    weak_records = []

    for page_input in pages:
        page_id = page_input["page_id"]
        numbering_payload = _load_json(ROOT / page_input["numbering_base"])
        image = cv2.imread(str(ROOT / page_input["image"]))
        if image is None:
            raise RuntimeError(f"Unable to read image: {page_input['image']}")
        h_img, w_img = image.shape[:2]
        current_overrides = []
        proposed_overrides = []

        for page_entry in numbering_payload.get("pages", []):
            page_number = page_entry.get("page_number", 1)
            for sys_idx, system in enumerate(page_entry.get("systems", [])):
                for m_idx, measure in enumerate(system.get("measures", [])):
                    x1, y1, x2, y2 = measure["bbox"]
                    margin = 20
                    cx1, cy1 = int(max(0, x1 - margin)), int(max(0, y1 - margin))
                    cx2, cy2 = int(min(w_img, x2 + margin)), int(min(h_img, y2 + margin))
                    crop = image[cy1:cy2, cx1:cx2]
                    prob = classifier.predict(crop)
                    if prob <= RESCUE_THRESHOLD:
                        continue
                    det = detect_with_probe(image, system, x1, y1, x2, y2, prob, w_img, h_img, ocr)
                    found_num = det["found_num"]
                    is_valid = False
                    status_label = ""
                    if found_num:
                        if prob > THRESHOLD:
                            is_valid, status_label = True, "found"
                        elif det["best_score"] > 60:
                            is_valid, status_label = True, "rescue"
                    if not is_valid:
                        continue
                    override = {
                        "page": page_number - 1,
                        "system": sys_idx,
                        "measure": m_idx,
                        "skip": found_num - 1,
                        "comment": f"CNN({prob:.2f})+OCR({det['best_score']:.1f}): {found_num}",
                    }
                    current_overrides.append(override)
                    rec = {
                        "page_id": page_id,
                        "key": [page_number - 1, sys_idx, m_idx],
                        "measure_number": measure.get("number"),
                        "found_num": found_num,
                        "skip": found_num - 1,
                        "prob": prob,
                        "ocr_score": det["best_score"],
                        "status": status_label,
                        "comment": override["comment"],
                        "strong_one_count": det["strong_one_count"],
                        "strong_one_max_conf": det["strong_one_max_conf"],
                        "proposed_veto": det["proposed_veto"],
                        "best_debug": det["best_debug"],
                    }
                    current_records.append(rec)
                    if prob < PROBE_PROB_MAX or det["best_score"] < PROBE_SCORE_MAX or det["strong_one_count"]:
                        weak_records.append(rec)
                    if det["proposed_veto"]:
                        vetoed_records.append(rec)
                    else:
                        proposed_overrides.append(override)

        current_summaries.append(build_eval_summary(page_input, numbering_payload, current_overrides))
        proposed_summaries.append(build_eval_summary(page_input, numbering_payload, proposed_overrides))

    current_agg = aggregate(current_summaries)
    proposed_agg = aggregate(proposed_summaries)
    detail = {
        "probe_rule": {
            "prob_must_be_below": PROBE_PROB_MAX,
            "ocr_score_must_be_below": PROBE_SCORE_MAX,
            "one_evidence_min_count": ONE_EVIDENCE_MIN_COUNT,
            "one_evidence_min_conf": ONE_EVIDENCE_MIN_CONF,
        },
        "current_aggregate": current_agg,
        "proposed_aggregate": proposed_agg,
        "vetoed_records": vetoed_records,
        "weak_or_one_evidence_records": weak_records,
        "current_record_count": len(current_records),
    }
    DETAIL_JSON.write_text(j(jsonable(detail)), encoding="utf-8")

    lines = []
    lines.append("# Issue #213 one-evidence guard probe")
    lines.append("")
    lines.append("## probe rule")
    lines.append(j(detail["probe_rule"]))
    lines.append("")
    lines.append("## current aggregate")
    lines.append(j(current_agg))
    lines.append("")
    lines.append("## proposed aggregate after veto")
    lines.append(j(proposed_agg))
    lines.append("")
    lines.append("## delta")
    lines.append(
        j(
            {
                k: proposed_agg[k] - current_agg[k]
                for k in [
                    "total_detected",
                    "matched_tp",
                    "missed_fn",
                    "skip_mismatch",
                    "unexpected_fp",
                ]
            }
        )
    )
    lines.append("")
    lines.append("## vetoed records")
    if vetoed_records:
        for rec in vetoed_records:
            lines.append(json.dumps(jsonable(rec), ensure_ascii=False))
    else:
        lines.append("<none>")
    lines.append("")
    lines.append("## weak / one-evidence current accepted records")
    for rec in weak_records:
        lines.append(json.dumps(jsonable(rec), ensure_ascii=False))
    lines.append("")
    lines.append(f"Full details: {DETAIL_JSON.relative_to(ROOT)}")

    REPORT.write_text("\n".join(lines), encoding="utf-8")
    print(REPORT.read_text(encoding="utf-8"))
    print("\nWrote: " + str(REPORT))


if __name__ == "__main__":
    main()
